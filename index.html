# safe_load_test.py # 用途：對「自己擁有或明確獲得授權」的網站進行受控壓力測試 # Python 3.10+ # # 安裝： # pip install requests # # 執行： # python safe_load_test.py import time import statistics import threading from concurrent.futures import ThreadPoolExecutor, as_completed from dataclasses import dataclass from urllib.parse import urlparse import requests # ========================= # 測試設定 # ========================= MAX_CONCURRENCY = 100 # 最大同時連線數，避免無限制發送 STEP = 10 # 每次增加多少併發 STEP_DURATION = 10 # 每個併發等級測試幾秒 REQUEST_TIMEOUT = 10 # 單次 Request 最長等待時間 # 如果網站出現大量錯誤，就停止繼續增加壓力 ERROR_RATE_STOP = 0.30 # 防止意外輸入非 HTTP/HTTPS ALLOWED_SCHEMES = {"http", "https"} @dataclass class RequestResult: success: bool status_code: int | None latency: float error: str | None = None def validate_url(url: str) -> str: """驗證測試網址。""" url = url.strip() if not url.startswith(("http://", "https://")): url = "https://" + url parsed = urlparse(url) if parsed.scheme not in ALLOWED_SCHEMES: raise ValueError("只允許 HTTP / HTTPS 網址。") if not parsed.netloc: raise ValueError("網址格式錯誤。") return url def send_request(url: str) -> RequestResult: """執行單次 GET Request。""" start = time.perf_counter() try: response = requests.get( url, timeout=REQUEST_TIMEOUT, headers={ "User-Agent": "ControlledLoadTester/1.0" }, ) latency = time.perf_counter() - start return RequestResult( success=200 <= response.status_code < 400, status_code=response.status_code, latency=latency, ) except requests.RequestException as exc: latency = time.perf_counter() - start return RequestResult( success=False, status_code=None, latency=latency, error=str(exc), ) def run_test(url: str, concurrency: int, duration: int): """執行指定併發數的壓力測試。""" results: list[RequestResult] = [] stop_event = threading.Event() def worker(): while not stop_event.is_set(): results.append(send_request(url)) print() print("=" * 70) print(f"開始測試：{concurrency} 個同時連線") print(f"測試時間：{duration} 秒") print("=" * 70) start_time = time.perf_counter() with ThreadPoolExecutor(max_workers=concurrency) as executor: futures = [ executor.submit(worker) for _ in range(concurrency) ] time.sleep(duration) stop_event.set() for future in as_completed(futures): try: future.result() except Exception: pass elapsed = time.perf_counter() - start_time if not results: return { "concurrency": concurrency, "requests": 0, "success": 0, "failed": 0, "rps": 0, "error_rate": 1, "avg_latency": 0, "p50": 0, "p95": 0, "p99": 0, "max_latency": 0, } latencies = sorted( result.latency for result in results ) success_count = sum( result.success for result in results ) failed_count = len(results) - success_count error_rate = failed_count / len(results) def percentile(values, percentile): index = int(len(values) * percentile / 100) if index >= len(values): index = len(values) - 1 return values[index] return { "concurrency": concurrency, "requests": len(results), "success": success_count, "failed": failed_count, "rps": len(results) / elapsed, "error_rate": error_rate, "avg_latency": statistics.mean(latencies), "p50": percentile(latencies, 50), "p95": percentile(latencies, 95), "p99": percentile(latencies, 99), "max_latency": max(latencies), } def print_result(result): """顯示測試結果。""" print() print(f"併發數 : {result['concurrency']}") print(f"總 Request : {result['requests']}") print(f"成功 : {result['success']}") print(f"失敗 : {result['failed']}") print(f"錯誤率 : {result['error_rate'] * 100:.2f}%") print(f"Requests/sec : {result['rps']:.2f}") print(f"平均延遲 : {result['avg_latency'] * 1000:.2f} ms") print(f"P50 延遲 : {result['p50'] * 1000:.2f} ms") print(f"P95 延遲 : {result['p95'] * 1000:.2f} ms") print(f"P99 延遲 : {result['p99'] * 1000:.2f} ms") print(f"最高延遲 : {result['max_latency'] * 1000:.2f} ms") def main(): print("=" * 70) print(" Controlled Website Load Tester") print("=" * 70) url = input("\n輸入你的網站網址：").strip() try: url = validate_url(url) except ValueError as exc: print(f"\n網址錯誤：{exc}") return print("\n測試目標：", url) print(f"最大併發：{MAX_CONCURRENCY}") print(f"每階段：{STEP_DURATION} 秒") confirm = input( "\n請確認這是你擁有或獲得授權測試的網站。輸入 YES 開始：" ).strip() if confirm != "YES": print("測試已取消。") return all_results = [] # 逐步增加壓力： # 10 → 20 → 30 → ... → MAX_CONCURRENCY for concurrency in range(STEP, MAX_CONCURRENCY + 1, STEP): result = run_test( url=url, concurrency=concurrency, duration=STEP_DURATION, ) all_results.append(result) print_result(result) # 錯誤率過高，停止增加壓力 if result["error_rate"] >= ERROR_RATE_STOP: print() print("=" * 70) print("偵測到大量 Request 失敗，停止增加壓力。") print("=" * 70) break if not all_results: return # 找出測試期間成功處理量最高的階段 best = max( all_results, key=lambda x: x["success"] ) print() print() print("#" * 70) print(" 測試總結") print("#" * 70) print(f""" 測試網址： {url} 最高成功 Request 數： {best['success']} 該階段總 Request： {best['requests']} 該階段併發數： {best['concurrency']} 該階段吞吐量： {best['rps']:.2f} Requests/sec 平均延遲： {best['avg_latency'] * 1000:.2f} ms P50： {best['p50'] * 1000:.2f} ms P95： {best['p95'] * 1000:.2f} ms P99： {best['p99'] * 1000:.2f} ms 最高延遲： {best['max_latency'] * 1000:.2f} ms 錯誤率： {best['error_rate'] * 100:.2f} % """) print("-" * 70) print("各階段統計") print("-" * 70) print( f"{'併發':>8}" f"{'Requests':>12}" f"{'成功':>12}" f"{'失敗':>12}" f"{'RPS':>12}" f"{'P95(ms)':>12}" f"{'錯誤率':>12}" ) for result in all_results: print( f"{result['concurrency']:>8}" f"{result['requests']:>12}" f"{result['success']:>12}" f"{result['failed']:>12}" f"{result['rps']:>12.2f}" f"{result['p95'] * 1000:>12.2f}" f"{result['error_rate'] * 100:>11.2f}%" ) print("#" * 70) print("測試完成") print("#" * 70) if __name__ == "__main__": main()
